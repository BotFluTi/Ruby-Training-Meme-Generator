$signupBody = @{
  user = @{
    username = 'burnetete'
    password = 'ananas'
  }
} | ConvertTo-Json -Depth 3

$signupResponse = Invoke-RestMethod `
  -Method Post `
  -Uri 'http://localhost:4567/signup' `
  -ContentType 'application/json' `
  -Body $signupBody

$token = $signupResponse.user.token
$token
